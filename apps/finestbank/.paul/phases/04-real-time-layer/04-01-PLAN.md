    ```python
    import json
    import redis.asyncio as aioredis
    from fastapi import FastAPI

    def channel_name(division_id: str) -> str:
        return f"transactions:{division_id}"

    async def init_redis(app: FastAPI, redis_url: str) -> None:
        app.state.redis = aioredis.from_url(redis_url, decode_responses=True)

    async def close_redis(app: FastAPI) -> None:
        await app.state.redis.aclose()

    async def publish_transaction(redis: aioredis.Redis, division_id: str, payload: dict) -> None:
        await redis.publish(channel_name(division_id), json.dumps(payload))
    ```

    Notes:
    - `decode_responses=True` — all messages come back as str, no manual decode needed
    - `aclose()` not `close()` — redis.asyncio uses async close
    - `publish_transaction` is the helper 04-03 will call from the simulator

<task type="auto">
  <name>Task 2: Create WebSocket endpoint ws.py</name>
  <files>backend/app/api/ws.py</files>
  <action>
    Create backend/app/api/ws.py:

    ```python
    import json
    import uuid

    import redis.asyncio as aioredis
    from fastapi import APIRouter, Query, WebSocket, WebSocketDisconnect
    from fastapi.websockets import WebSocketState

    from app.core.auth import decode_jwt
    from app.core.redis_client import channel_name

    router = APIRouter(tags=["websocket"])

    @router.websocket("/ws/live/{division_id}")
    async def live_feed(
        websocket: WebSocket,
        division_id: str,
        token: str = Query(...),
    ):
        # Validate JWT before accepting
        try:
            payload = decode_jwt(token)
            role = payload.get("role", "")
            if role not in ("cfo", "analyst"):
                await websocket.close(code=1008)
                return
        except Exception:
            await websocket.close(code=1008)
            return

        await websocket.accept()

        redis: aioredis.Redis = websocket.app.state.redis
        channel = channel_name(division_id)
        pubsub = redis.pubsub()
        await pubsub.subscribe(channel)

        try:
            async for message in pubsub.listen():
                if websocket.client_state != WebSocketState.CONNECTED:
                    break
                if message["type"] == "message":
                    await websocket.send_text(message["data"])
        except WebSocketDisconnect:
            pass
        finally:
            await pubsub.unsubscribe(channel)
            await pubsub.aclose()
    ```

    Notes:
    - Token via query param — WebSocket upgrade cannot send custom headers in browser
    - `decode_jwt` already exists in app/core/auth.py (used by deps.py); reuse it
    - `websocket.app.state.redis` accesses the pool stored by lifespan
    - Catch WebSocketDisconnect only — other exceptions bubble up to FastAPI handler
    - `WebSocketState.CONNECTED` guard prevents send after disconnect
    - Do NOT import get_scoped_db or RLS — WS endpoint doesn't query DB directly; it only receives published events
